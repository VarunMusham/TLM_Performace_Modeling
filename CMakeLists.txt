cmake_minimum_required(VERSION 3.16)
project(TLMPerformanceLab LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TLM_LAB_ALLOW_UNQUALIFIED_SYSTEMC "Allow a non-target SystemC for local smoke checks" OFF)
find_package(SystemCLanguage CONFIG REQUIRED)
if(SystemCLanguage_VERSION VERSION_LESS "2.3.4" OR SystemCLanguage_VERSION VERSION_GREATER_EQUAL "2.3.5")
    if(NOT TLM_LAB_ALLOW_UNQUALIFIED_SYSTEMC)
        message(FATAL_ERROR "Target is SystemC 2.3.4; found ${SystemCLanguage_VERSION}. See docs/SETUP_WINDOWS.md.")
    endif()
    message(WARNING "UNQUALIFIED SystemC ${SystemCLanguage_VERSION}; target 2.3.4 validation remains pending.")
endif()

add_executable(wdt_pilot examples/wdt/wdt.cpp examples/wdt/testbench.cpp)
target_link_libraries(wdt_pilot PRIVATE SystemC::systemc)
target_compile_definitions(wdt_pilot PRIVATE SC_INCLUDE_DYNAMIC_PROCESSES)
if(MSVC)
    target_compile_options(wdt_pilot PRIVATE /W4)
else()
    target_compile_options(wdt_pilot PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/traces")
foreach(scenario expiry kick_boundary disable_boundary reset_pending invalid annotated_delay read_counter peq_on_edge)
    add_test(NAME wdt_${scenario} COMMAND wdt_pilot ${scenario} "${CMAKE_CURRENT_BINARY_DIR}/traces/${scenario}.csv")
    set_tests_properties(wdt_${scenario} PROPERTIES TIMEOUT 10)
    add_test(NAME wdt_${scenario}_no_trace COMMAND wdt_pilot ${scenario} -)
    set_tests_properties(wdt_${scenario}_no_trace PROPERTIES TIMEOUT 10)
endforeach()
