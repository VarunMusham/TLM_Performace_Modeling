cmake_minimum_required(VERSION 3.16)
project(CustomerIPExample LANGUAGES CXX)
add_executable(ip_demo testbench.cpp)
target_compile_features(ip_demo PRIVATE cxx_std_14)

    set_target_properties(ip_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
